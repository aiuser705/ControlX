export const BackgroundShader = {
  uniforms: {
    uTime: { value: 0.0 },
    uResolution: { value: [1920, 1080] },
    uMouse: { value: [0.5, 0.5] },
    uColorStart: { value: [0.98, 0.97, 0.96] }, // #FAF9F5
    uColorEnd: { value: [0.88, 0.87, 0.83] }     // #E2DFD5
  },
  vertexShader: /* glsl */ `
    varying vec2 vUv;
    void main() {
      vUv = uv;
      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    }
  `,
  fragmentShader: /* glsl */ `
    uniform float uTime;
    uniform vec2 uResolution;
    uniform vec2 uMouse;
    uniform vec3 uColorStart;
    uniform vec3 uColorEnd;
    varying vec2 vUv;

    void main() {
      vec2 st = gl_FragCoord.xy / uResolution.xy;
      vec2 mouseEffect = uMouse * 0.03;
      
      float wave = sin(st.x * 3.5 + uTime * 0.4 + mouseEffect.x) * 
                   cos(st.y * 3.5 + uTime * 0.4 + mouseEffect.y) * 0.04;
                   
      vec3 finalColor = mix(uColorStart, uColorEnd, st.y + wave);
      gl_FragColor = vec4(finalColor, 1.0);
    }
  `
};

export const ParticleShader = {
  uniforms: {
    uTime: { value: 0.0 },
    uProgress: { value: 0.0 },
    uBaseColor: { value: [0.058, 0.509, 0.349] }, // #0F8259
    uHighlightColor: { value: [0.062, 0.725, 0.505] } // #10B981
  },
  vertexShader: /* glsl */ `
    uniform float uTime;
    uniform float uProgress;
    attribute vec3 aVelocity;
    attribute float aScale;
    varying float vAlpha;
    varying vec3 vColor;
    
    uniform vec3 uBaseColor;
    uniform vec3 uHighlightColor;

    void main() {
      vec3 pos = position;
      
      // Spiral trajectory calculation
      float angle = uTime * 1.5 + pos.y * 0.5;
      float radius = length(pos.xz) + sin(uTime * 2.0 + pos.y) * 0.2;
      
      pos.x = cos(angle) * radius + aVelocity.x * uProgress * 2.0;
      pos.z = sin(angle) * radius + aVelocity.z * uProgress * 2.0;
      pos.y += aVelocity.y * uProgress * 1.5;

      vec4 mvPosition = modelViewMatrix * vec4(pos, 1.0);
      gl_PointSize = (aScale * 12.0) * (1.0 / -mvPosition.z) * (1.0 + uProgress);
      gl_Position = projectionMatrix * mvPosition;

      vAlpha = smoothstep(0.0, 0.2, uProgress) * (1.0 - smoothstep(0.8, 1.0, uProgress));
      vColor = mix(uBaseColor, uHighlightColor, sin(pos.y + uTime) * 0.5 + 0.5);
    }
  `,
  fragmentShader: /* glsl */ `
    varying float vAlpha;
    varying vec3 vColor;

    void main() {
      float dist = length(gl_PointCoord - vec2(0.5));
      if (dist > 0.5) discard;
      
      float softEdge = smoothstep(0.5, 0.0, dist);
      gl_FragColor = vec4(vColor, vAlpha * softEdge);
    }
  `
};

"use client";

import React, { useRef } from "react";
import { useFrame } from "@react-three/fiber";
import * as THREE from "three";

interface GlassXMeshProps {
  progress: number;
}

export const GlassXMesh: React.FC<GlassXMeshProps> = ({ progress }) => {
  const meshRef = useRef<THREE.Group>(null);
  const materialRef = useRef<THREE.MeshPhysicalMaterial>(null);

  useFrame((state, delta) => {
    if (meshRef.current) {
      // Idle float oscillation
      meshRef.current.position.y = Math.sin(state.clock.elapsedTime * 0.8) * 0.08;
      meshRef.current.rotation.y = THREE.MathUtils.lerp(
        meshRef.current.rotation.y,
        state.pointer.x * 0.3,
        0.05
      );
      meshRef.current.rotation.x = THREE.MathUtils.lerp(
        meshRef.current.rotation.x,
        -state.pointer.y * 0.2,
        0.05
      );
    }
  });

  return (
    <group ref={meshRef} position={[0, 0, 0]} scale={[1.2, 1.2, 1.2]}>
      {/* Primary Polyhedral Glass X Body */}
      <mesh>
        <boxGeometry args={[0.8, 2.8, 0.4]} />
        <meshPhysicalMaterial
          ref={materialRef}
          color="#0F8259"
          transmission={0.95}
          opacity={1.0}
          transparent={true}
          roughness={0.05}
          metalness={0.0}
          ior={1.52}
          thickness={0.8}
          attenuationColor="#0F8259"
          attenuationDistance={0.5}
          clearcoat={1.0}
          clearcoatRoughness={0.1}
          reflectivity={0.9}
        />
      </mesh>
      <mesh rotation={[0, 0, Math.PI / 2]}>
        <boxGeometry args={[0.8, 2.8, 0.4]} />

      </mesh>
    </group>
  );
};

"use client";

import React, { useMemo } from "react";
import * as THREE from "three";

export const ProcessRail: React.FC = () => {
  const curve = useMemo(() => {
    return new THREE.CatmullRomCurve3([
      new THREE.Vector3(-4, 0, 0),
      new THREE.Vector3(-2, 0.5, 0.5),
      new THREE.Vector3(0, -0.2, 0),
      new THREE.Vector3(2, 0.4, -0.5),
      new THREE.Vector3(4, 0, 0)
    ]);
  }, []);

  const tubeGeometry = useMemo(() => {
    return new THREE.TubeGeometry(curve, 64, 0.08, 16, false);
  }, [curve]);

  return (
    <mesh geometry={tubeGeometry}>
      <meshPhysicalMaterial
        color="#0F8259"
        transmission={0.92}
        roughness={0.08}
        ior={1.48}
        thickness={0.3}
        transparent={true}
        opacity={0.85}
      />
    </mesh>
  );
};

import React, { useEffect, useRef } from "react";
import { Canvas } from "@react-three/fiber";
import Lenis from "@studio-freight/lenis";
import { gsap } from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";

import { GlassXMesh } from "@/components/canvas/GlassXMesh";
import { ProcessRail } from "@/components/canvas/ProcessRail";
import { Navbar } from "@/components/dom/Navbar";
import { ServicesGrid } from "@/components/dom/ServicesGrid";
import { ContactForm } from "@/components/dom/ContactForm";

gsap.registerPlugin(ScrollTrigger);

export default function Home() {
  const mainRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    // Initialize Lenis Smooth Scroll Engine
    const lenis = new Lenis({
      duration: 1.2,
      easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
      orientation: "vertical",
      smoothWheel: true
    });

    lenis.on("scroll", ScrollTrigger.update);

    gsap.ticker.add((time) => {
      lenis.raf(time * 1000);
    });

    gsap.ticker.lagSmoothing(0);

    // ScrollTrigger Choreography Pipeline
    const ctx = gsap.context(() => {
      gsap.to(".hero-title", {
        scrollTrigger: {
          trigger: ".sec-hero",
          start: "top top",
          end: "bottom top",
          scrub: true
        },
        y: -100,
        opacity: 0
      });
    }, mainRef);

    return () => {
      lenis.destroy();
      ctx.revert();
    };
  }, []);

  return (
    <main ref={mainRef} className="relative bg-[#EBE9E1] text-[#18191C] min-h-screen selection:bg-[#0F8259] selection:text-white">
      {/* WebGL Experience Layer */}
      <div className="fixed inset-0 z-0 pointer-events-none">
        <Canvas
          camera={{ position: [0, 0, 8], fov: 45, near: 0.1, far: 1000 }}
          gl={{ antialias: true, alpha: true, powerPreference: "high-performance" }}
          dpr={[1, 2]}

          <ambientLight intensity={0.75} />
          <directionalLight position={[5, 10, 7]} intensity={1.2} />
          <pointLight position={[0, -1, 2]} intensity={0.8} color="#0F8259" />
          
          <GlassXMesh progress={0} />
          <ProcessRail />
        </Canvas>
      </div>

      {/* Interactive DOM Layer */}
      <div className="relative z-10">
        <Navbar />

        {/* Hero Section */}
        <section className="sec-hero h-screen flex flex-col justify-center items-center text-center px-6">
          <h1 className="hero-title font-serif text-6xl md:text-7xl font-semibold tracking-tight max-w-4xl leading-tight">
